A working answer to the configurator question, built so it runs anywhere without a plugin, a library or a build step. Try it below, then take the code.

Try it

Choose a color. Upload any logo file from this computer and drag it where you want it. Nothing is uploaded anywhere; the file never leaves your browser.

Your logo

Choose a color. Upload a logo and drag it where you want it.

What it still needs: one photograph. The sixteen color files on the bench page are flat illustrations at 209 by 143 pixels, made as swatches for the color grid, and there is nothing larger on the server. The preview is capped at a size where they hold up, but they will not carry a full-width feature. One photograph of a plain white bench with no graphics on it, shot square-on from the side, produces all sixteen colors from that single image and lets a customer's logo sit on the real surface. There is a switch at the top of the code for it.

Take the code

Paste the whole thing into an Elementor HTML widget. The three hidden fields at the bottom carry the color, the file name and the placement, so the quote form collects all three with the request.

<!-- ===================================================================
     EasyRec bench preview: pick a color, drop a logo on it.

     TO USE: paste everything from <div id="erb"> down to the last
     </script> into an Elementor HTML widget. Nothing else needed,
     no plugin, no library, no build step.

     You can also just double-click this file to see it run in a browser.

     The 16 photos are the ones already on
     easyrecreation.com/easy-stack-bench/, loaded straight from your
     media library. Swap any URL below and the swatch follows it.

     ONE THING TO KNOW: those 16 files are flat illustrations of the
     bench, 209 x 143 pixels, and nothing larger exists on the server.
     They were made as little color-grid swatches, which is all the page
     uses them for today. So this is capped at a size where they still
     hold up, rather than blown up into mush.

     The upgrade is one photograph: a plain white bench with no graphics
     on it, shot square-on from the side. Put its URL in PHOTO below and
     set USE_PHOTO to true, and all 16 colors come off that single image
     with the logo sitting on the real surface.
     =================================================================== -->

<script>
(function () {
  var BASE = 'https://easyrecreation.com/wp-content/uploads/';

  // Flip this to true once you have the white-bench photo, and put its
  // URL in PHOTO. Every color is then produced from that one image.
  var USE_PHOTO = false;
  var PHOTO = '';

  // Name, RAL, swatch color, photo. Pairing taken from the live
  // easy-stack-bench page. The filenames do not always match the RAL
  // code, which is their filing, not a mistake here.
  var COLORS = [
    ['Traffic Red',    'RAL 3020', '#a72920', '2020/03/EasyStackBench_3002.jpg'],
    ['Wine Red',       'RAL 3005', '#5e2129', '2020/03/EasyStackBench_3005.jpg'],
    ['Orange',         'RAL 2004', '#e25303', '2020/03/EasyStackBench_2008.jpg'],
    ['Golden Yellow',  'RAL 1004', '#e2b007', '2024/02/EasyStackBench_1023-golden-yellow.jpg'],
    ['Traffic Yellow', 'RAL 1023', '#f7ba0b', '2024/02/EasyStackBench_1023-trafficyellow-01.jpg'],
    ['Traffic Green',  'RAL 6024', '#308446', '2020/03/EasyStackBench_6024-6032.jpg'],
    ['Fir Green',      'RAL 6009', '#27352a', '2020/03/EasyStackBench_6005.jpg'],
    ['Light Blue',     'RAL 5012', '#0089b6', '2024/02/EasyStackBench_light-blue-01.jpg'],
    ['Signal Blue',    'RAL 5005', '#005387', '2020/03/EasyStackBench_5005.jpg'],
    ['Cobalt Blue',    'RAL 5013', '#1e213d', '2020/03/EasyStackBench_5003.jpg'],
    ['Terra Brown',    'RAL 8028', '#4e3b31', '2020/03/EasyStackBench_8028.jpg'],
    ['Traffic Black',  'RAL 9017', '#1e1e1e', '2020/03/EasyStackBench_9017.jpg'],
    ['Iron Gray',      'RAL 7011', '#52595d', '2020/03/EasyStackBench_7011.jpg'],
    ['Cool Gray',      'RAL 7040', '#9da3a6', '2020/03/EasyStackBench_7040.jpg'],
    ['Silk Gray',      'RAL 7044', '#cac4b0', '2020/03/EasyStackBench_7044.jpg'],
    ['White',          'RAL 9016', '#f1f0ea', '2020/03/EasyStackBench_9016.jpg']
  ];

  var root    = document.getElementById('erb');
  var stage   = root.querySelector('.erb-stage');
  var bench   = document.getElementById('erb-bench');
  var logo    = document.getElementById('erb-logo');
  var logoImg = document.getElementById('erb-logo-img');
  var size    = document.getElementById('erb-size');
  var clear   = document.getElementById('erb-clear');
  var hint    = document.getElementById('erb-hint');
  var summary = document.getElementById('erb-summary');
  var outColor = document.getElementById('erb-out-color');
  var outFile  = document.getElementById('erb-out-file');
  var outPos   = document.getElementById('erb-out-pos');

  var chosen = null, filename = '', pos = { x: 50, y: 38 };

  COLORS.forEach(function (c, i) {
    var b = document.createElement('button');
    b.type = 'button';
    b.className = 'erb-sw';
    b.style.background = c[2];
    b.title = c[0] + ' ' + c[1];
    b.setAttribute('aria-label', c[0] + ' ' + c[1]);
    b.setAttribute('aria-pressed', 'false');
    b.addEventListener('click', function () { pick(i); });
    document.getElementById('erb-swatches').appendChild(b);
  });

  function pick(i) {
    chosen = COLORS[i];
    if (USE_PHOTO && PHOTO) {
      bench.src = PHOTO;
      bench.style.mixBlendMode = 'multiply';
      stage.style.background = chosen[2];
    } else {
      bench.src = BASE + chosen[3];
    }
    bench.alt = 'Easy Stack Bench in ' + chosen[0] + ' ' + chosen[1];
    Array.prototype.forEach.call(root.querySelectorAll('.erb-sw'), function (b, n) {
      b.setAttribute('aria-pressed', n === i ? 'true' : 'false');
    });
    write();
  }

  document.getElementById('erb-file').addEventListener('change', function (e) {
    var f = e.target.files && e.target.files[0];
    if (!f) return;
    filename = f.name;
    var r = new FileReader();
    r.onload = function (ev) {
      logoImg.src = ev.target.result;
      logo.hidden = false;
      size.disabled = false;
      clear.hidden = false;
      hint.textContent = 'Drag the logo to move it. Use the slider to size it.';
      write();
    };
    r.readAsDataURL(f);
  });

  size.addEventListener('input', function () {
    logo.style.width = size.value + '%';
    write();
  });

  clear.addEventListener('click', function () {
    logo.hidden = true;
    size.disabled = true;
    clear.hidden = true;
    filename = '';
    logoImg.src = '';
    document.getElementById('erb-file').value = '';
    hint.textContent = 'Choose a color. Upload a logo and drag it where you want it.';
    write();
  });

  var dragging = false;
  logo.addEventListener('pointerdown', function (e) {
    dragging = true;
    logo.classList.add('erb-dragging');
    logo.setPointerCapture(e.pointerId);
  });
  logo.addEventListener('pointermove', function (e) {
    if (!dragging) return;
    var r = stage.getBoundingClientRect();
    pos.x = Math.min(95, Math.max(5, ((e.clientX - r.left) / r.width) * 100));
    pos.y = Math.min(95, Math.max(5, ((e.clientY - r.top) / r.height) * 100));
    logo.style.left = pos.x + '%';
    logo.style.top  = pos.y + '%';
    write();
  });
  logo.addEventListener('pointerup', function (e) {
    dragging = false;
    logo.classList.remove('erb-dragging');
    logo.releasePointerCapture(e.pointerId);
  });

  function write() {
    outColor.value = chosen ? chosen[0] + ' (' + chosen[1] + ')' : '';
    outFile.value  = filename;
    outPos.value   = filename
      ? Math.round(pos.x) + '% across, ' + Math.round(pos.y) + '% down, ' + size.value + '% wide'
      : '';
    var parts = [];
    if (chosen) parts.push(chosen[0] + ' ' + chosen[1]);
    if (filename) parts.push('logo: ' + filename);
    summary.textContent = parts.join('  |  ');
  }

  pick(8); // opens on Signal Blue
})();
</script>
